    'use strict';

// imports
import { AxiosInstance, AxiosRequestConfig, AxiosResponse } from 'axios';
import qs from 'qs'

class Dao1 {

  private axios: AxiosInstance;
  private sessionCookieName: string;
  private sessionCookie: string;

  // constructor
  constructor(axios: AxiosInstance) {
    // axios library used to make the HTTP requests
    this.axios = axios;
    // session cookie
    this.sessionCookieName = "PHPSESSID";
    this.sessionCookie = '';
  }

  // init session
  async initSession(): Promise<any> {
    // options of the HTTP request [get /main.php?action=init-session&type=json]
    const options: AxiosRequestConfig = {
      // [TypeScript update] URL included in [options.url] (see getRemoteData)
      url: 'main.php',
      method: "GET",
      // URL parameters
      params: {
        action: 'init-session',
        type: 'json'
      }
    };
    // executing the HTTP request
    return await this.getRemoteData(options);
  }

  async authenticateUser(user: string, password: string): Promise<any> {
    // options of the HTTP request [post /main.php?action=authenticate-user]
    const options: AxiosRequestConfig = {
      url: 'main.php',
      method: "POST",
      headers: {
        'Content-type': 'application/x-www-form-urlencoded',
      },
      // body of the POST
      data: qs.stringify({
        user: user,
        password: password
      }),
      // URL parameters
      params: {
        action: 'authenticate-user'
      }
    };
    // executing the HTTP request
    return await this.getRemoteData(options);
  }

  // tax calculation
  async calculateTax(married: string, children: number, salary: number): Promise<any> {
    // options of the HTTP request
    const options: AxiosRequestConfig = {
      url: 'main.php',
      method: "POST",
      headers: {
        'Content-type': 'application/x-www-form-urlencoded',
      },
      // body of the POST
      data: qs.stringify({
        married: married,
        children: children,
        salary: salary
      }),
      // URL parameters
      params: {
        action: 'calculate-tax'
      }
    };
    // executing the HTTP request
    const data = await this.getRemoteData(options);
    // result
    return data;
  }

  // list of simulations
  async listSimulations(): Promise<any> {
    // options of the HTTP request
    const options: AxiosRequestConfig = {
      url: 'main.php',
      method: "GET",
      // URL parameters
      params: {
        action: 'list-simulations'
      },
    };
    // executing the HTTP request
    const data = await this.getRemoteData(options);
    // result
    return data;
  }

  // delete a simulation
  async deleteSimulation(index: number): Promise<any> {
    // options of the HTTP request
    const options: AxiosRequestConfig = {
      url: 'main.php',
      method: "GET",
      // URL parameters
      params: {
        action: 'delete-simulation',
        number: index
      },
    };
    // executing the HTTP request
    const data = await this.getRemoteData(options);
    // result
    return data;
  }

  async getRemoteData(options: AxiosRequestConfig): Promise<any> {
    // for the session cookie
    if (!options.headers) {
      options.headers = {} as any;
    }
    (options.headers as any).Cookie = this.sessionCookie;
    // executing the HTTP request
    let response: AxiosResponse;
    try {
      // asynchronous request
      response = await this.axios.request(options);
    } catch (error: any) {
      // the [error] parameter is an exception instance - it can take various forms
      if (error.response) {
        // the server's response is in [error.response]
        response = error.response;
      } else {
        // we rethrow the error
        throw error;
      }
    }
    // response is the whole HTTP response of the server (HTTP headers + the response itself)
    // we retrieve the session cookie if it exists
    const setCookie = response.headers['set-cookie'];
    if (setCookie) {
      // setCookie is an array
      // we look for the session cookie in this array
      let found = false;
      let i = 0;
      while (!found && i < setCookie.length) {
        // we look for the session cookie
        const results = RegExp('^(' + this.sessionCookieName + '.+?);').exec(setCookie[i]);
        if (results) {
          // we store the session cookie
          // eslint-disable-next-line require-atomic-updates
          this.sessionCookie = results[1];
          // we found it
          found = true;
        } else {
          // next element
          i++;
        }
      }
    }
    // the server's response is in [response.data]
    return response.data;
  }
}

// exporting the class
export default Dao1;

// asynchronous [main] function
async function main(): Promise<void> {
  // axios configuration
  axios.defaults.timeout = 2000;
  axios.defaults.baseURL = 'http://localhost:3000';
  // instantiating the [dao] layer
  const dao = new Dao(axios);
  // using the [dao] layer
  try {
    // init session

    let response = await dao.initSession();

    // authentication
    log("-----------authenticate-user");
    response = await dao.authenticateUser("admin", "admin");
    log(response);
    // tax calculations
    log("-----------calculate-tax x 3");
    response = await Promise.all([
      dao.calculateTax("yes", 2, 45000),
      dao.calculateTax("no", 2, 45000),

    log(response);
    // list of simulations
    log("-----------list-of-simulations");
    response = await dao.listSimulations();
    log(response);
    // deleting a simulation
    log("-----------deleting simulation no. 1");
    response = await dao.deleteSimulation(1);
    log(response);
  } catch (error: any) {
    // we log the error
    console.log("error=", error.message);
  }
}

// Business class
class Business {

  // [any]: the exact structure of the data sent by the tax server (rates, ceilings, ...)
  // is not documented here; so we deliberately stay flexible on this type
  taxAdminData: any;

  // constructor
  constructor(taxAdmindata: any) {
    // this.taxAdminData: tax administration data
    this.taxAdminData = taxAdmindata;
  }

  // tax calculation
  // --------------------------------------------------------------------------
  calculateTax(married: string, children: number, salary: number): any {
    // married: yes, no
    // children: number of children
    // salary: annual salary
    // this.taxAdminData: tax administration data
    //
    // tax calculation with children
    const result1 = this.calculateTax2(married, children, salary);
    const tax1 = result1["tax"];
    // tax calculation without the children
    let result2, tax2, halfShareCeiling;
    if (children !== 0) {
      result2 = this.calculateTax2(married, 0, salary);
      tax2 = result2["tax"];
      // applying the family quotient cap
      halfShareCeiling = this.taxAdminData.halfShareQfCeiling;
      if (children < 3) {
        // HALF_SHARE_QF_CEILING euros for the first 2 children
        tax2 = tax2 - children * halfShareCeiling;
      } else {
        // HALF_SHARE_QF_CEILING euros for the first 2 children, double for the following ones
        tax2 = tax2 - 2 * halfShareCeiling - (children - 2) * 2 * halfShareCeiling;
      }
    } else {
      // no recalculation of the tax
      tax2 = tax1;
      result2 = result1;
    }
    // we take the higher tax in [tax1, tax2]
    let tax, rate, surcharge;
    if (tax1 > tax2) {
      tax = tax1;
      rate = result1["rate"];
      surcharge = result1["surcharge"];
    } else {
      surcharge = tax2 - tax1 + result2["surcharge"];
      tax = tax2;
      rate = result2["rate"];
    }
    // calculation of a possible discount
    const discount = this.getDiscount(married, tax);
    tax -= discount;
    // calculation of a possible tax reduction
    const reduction = this.getReduction(married, salary, children, tax);
    tax -= reduction;
    // result
    return {
      "tax": Math.floor(tax), "surcharge": surcharge, "discount": discount, "reduction": reduction,
      "rate": rate
    };
  }

  // --------------------------------------------------------------------------
  calculateTax2(married: string, children: number, salary: number): any {
    // married: yes, no
    // children: number of children
    // salary: annual salary
    // this->taxAdminData: tax administration data
    // 
    // number of shares
    married = married.toLowerCase();
    let shares;
    if (married === "yes") {
      shares = children / 2 + 2;
    } else {
      shares = children / 2 + 1;
    }
    // 1 share per child from the 3rd one
    if (children >= 3) {
      // one more half-share for each child from the 3rd one
      shares += 0.5 * (children - 2);
    }
    // taxable income
    const taxableIncome = this.getTaxableIncome(salary);
    // surcharge
    let surcharge = Math.floor(taxableIncome - 0.9 * salary);
    // for rounding issues
    if (surcharge < 0) {
      surcharge = 0;
    }
    // family quotient
    const quotient = taxableIncome / shares;
    // tax calculation
    const limits = this.taxAdminData.limits;
    const coeffR = this.taxAdminData.coeffR;
    const coeffN = this.taxAdminData.coeffN;
    // is put at the end of the limits array to stop the following loop
    limits[limits.length - 1] = quotient;
    // looking for the tax rate
    let i = 0;
    while (quotient > limits[i]) {
      i++;
    }
    // because we placed quotient at the end of the limits array, the previous loop
    // cannot go beyond the limits array
    // now we can calculate the tax
    const tax = Math.floor(taxableIncome * coeffR[i] - shares * coeffN[i]);
    // result
    return { "tax": tax, "surcharge": surcharge, "rate": coeffR[i] };
  }

  // taxableIncome=annualSalary-allowance
  // the allowance has a min and a max
  getTaxableIncome(salary: number): number {
    // allowance of 10% of the salary
    let allowance = 0.1 * salary;
    // this allowance cannot exceed taxAdminData.getTenPercentAllowanceMax()
    if (allowance > this.taxAdminData.tenPercentAllowanceMax) {
      allowance = this.taxAdminData.tenPercentAllowanceMax;
    }
    // the allowance cannot be lower than taxAdminData.getTenPercentAllowanceMin()
    if (allowance < this.taxAdminData.tenPercentAllowanceMin) {
      allowance = this.taxAdminData.tenPercentAllowanceMin;
    }
    // taxable income
    const taxableIncome = salary - allowance;
    // result
    return Math.floor(taxableIncome);
  }

  // calculates a possible discount
  getDiscount(married: string, taxes: number): number {
    // initially, a zero discount
    let discount = 0;
    // maximum tax amount to get the discount
    let taxCeilingForDiscount = married === "yes" ?
      this.taxAdminData.coupleTaxCeilingForDiscount :
      this.taxAdminData.singleTaxCeilingForDiscount;
    let discountCeiling;
    if (taxes < taxCeilingForDiscount) {
      // maximum amount of the discount
      discountCeiling = married === "yes" ?
        this.taxAdminData.coupleDiscountCeiling :
        this.taxAdminData.singleDiscountCeiling;
      // theoretical discount
      discount = discountCeiling - 0.75 * taxes;
      // the discount cannot exceed the amount of the tax
      if (discount > taxes) {
        discount = taxes;
      }
      // no discount <0
      if (discount < 0) {
        discount = 0;
      }
    }
    // result
    return Math.ceil(discount);
  }

  // calculates a possible reduction
  getReduction(married: string, salary: number, children: number, taxes: number): number {
    // the income ceiling to be entitled to the 20% reduction
    let incomeCeilingForReduction = married === "yes" ?
      this.taxAdminData.coupleIncomeCeilingForReduction :
      this.taxAdminData.singleIncomeCeilingForReduction;

    }
    // taxable income
    const taxableIncome = this.getTaxableIncome(salary);
    // reduction
    let reduction = 0;
    if (taxableIncome < incomeCeilingForReduction) {
      // 20% reduction
      reduction = 0.2 * taxes;
    }
    // result

    // options of the HTTP request  [get /main.php?action=get-admindata]
    const options: AxiosRequestConfig = {
      url: 'main.php',
      method: "GET",
      // URL parameters
      params: {
        action: 'get-admindata'
      }
    };
    // executing the HTTP request
    const data = await this.getRemoteData(options);
    // result
    return data;

  async getRemoteData(options: AxiosRequestConfig): Promise<any> {
    // for the session cookie
    if (!options.headers) {
      options.headers = {} as any;
    }
    (options.headers as any).Cookie = this.sessionCookie;
    // executing the HTTP request
    let response: AxiosResponse;
    try {
      // asynchronous request
      response = await this.axios.request(options);
    } catch (error: any) {
      // the [error] parameter is an exception instance - it can take various forms
      if (error.response) {
        // the server's response is in [error.response]
        response = error.response;
      } else {
        // we rethrow the error
        throw error;
      }
    }
    // response is the whole HTTP response of the server (HTTP headers + the response itself)
    // we retrieve the session cookie if it exists
    const setCookie = response.headers['set-cookie'];
    if (setCookie) {
      // setCookie is an array
      // we look for the session cookie in this array
      let found = false;
      let i = 0;
      while (!found && i < setCookie.length) {
        // we look for the session cookie
        const results = RegExp('^(' + this.sessionCookieName + '.+?);').exec(setCookie[i]);
        if (results) {
          // we store the session cookie
          // eslint-disable-next-line require-atomic-updates
          this.sessionCookie = results[1];
          // we found it
          found = true;
        } else {
          // next element
          i++;
        }
      }
    }
    // the server's response is in [response.data]
    return response.data;
  }

      }

      url: 'main.php',
      method: "GET",
      // URL parameters
      params: {
        action: 'get-admindata'
      }
    };
    // executing the HTTP request
    const data = await this.getRemoteData(options);
    // result
    return data;
  }

  async getRemoteData(options: AxiosRequestConfig): Promise<any> {
    // executing the HTTP request
    let response: AxiosResponse;
    try {
      // asynchronous request
      response = await this.axios.request(options);
    } catch (error: any) {
      // the [error] parameter is an exception instance - it can take various forms
      if (error.response) {
        // the server's response is in [error.response]
        response = error.response;
      } else {
        // we rethrow the error
        throw error;
      }
    }

    // the server's response is in [response.data]

    6.5.1. Solution Architecture

    This client, HTML, operates exactly the same way as [main3.ts], and this is a key design choice: it only calls the server (PHP) for three actions—always the same ones: init-session, authenticate-user, and get-admindata. However, the server’s code (PHP) shows that it is also capable of calculating the tax itself and storing the simulations in its session (this is what the actions calculate-tax, list-simulations, and delete-simulation, found in the [Controllers] folder)—but this client does not use them.

    The reason is simple: once the tax data (rates, thresholds, etc.) has been retrieved just once using get-admindata, everything needed to calculate the tax is already in the browser, thanks to the [Business] class. Requesting this calculation from the server for each simulation would require an avoidable round trip over the network; likewise, since each calculated simulation is immediately stored in the browser, requesting the list again from the server (list-simulations) would add no value. This client therefore minimizes server requests as much as possible—exactly the principle already implemented in [main3.ts]:

    • the [dao] layer ([Dao3.ts]) remains unchanged: its three methods already presented above (initSession, authenticateUser, getAdminData) are sufficient; none are added to it;
    • immediately after successful authentication, getAdminData is called once and for all, and its result is used to construct a Business object (new Business(taxAdminData)), which is retained throughout the entire work session;
    • each tax calculation then directly calls business.calculateTax(married, children, salary)—an ordinary JavaScript function, executed in the browser, without any network calls;
    • the calculated simulations are stored in a simple array JavaScript, kept in memory by the controller—one push per calculation, one splice per deletion, exactly as [main3.ts] does with its instructions simulations.push(...) and simulations.splice(1, 1).

    A slight difference from [main3.ts]: the business.calculateTax method returns only the values it has calculated (tax, surcharge, discount, reduction, rate), not the parameters entered by the user (married, children, salary). [main3.ts] doesn’t need them, since it simply uses console.log; but our view [List of simulations] must display these three parameters in the table—so the controller attaches them to the result itself before storing it (we’ll see this in the code for [AppHtml.ts]).

    Finally, the [End of session] view no longer makes a call to the server either: it simply discards the [Business] object and the list of simulations, then redisplays the authentication screen. Recall that the responses from the three server actions still in use always have the same format, which we’ve already seen in [main3.ts]: a JSON object { action, status, response }, where response contains either the requested result (if successful) or an error message—a string or an array of strings, depending on the relevant PHP controller.

    6.5.1.5. The controller: the [AppHtml.ts] file

    This is the application’s orchestrator: it takes on the role played by the main() function of **[main3.ts], but in the form of a class, with one method per view. It now coordinates two layers, [dao] (the server) and [business] (the calculation), before instructing the [views]** layer to display either a result or an error message:

    
// imports
import axios from 'axios';
import "core-js/stable";
import "regenerator-runtime/runtime";

// imports
import Dao from './Dao3';
import Business from './Business';
import Views, { Simulation } from './Views';

// -----------------------------------------------------------------------
// controller of the HTML application: orchestrates the calls to the server
// ([dao] layer), the calculations ([business] layer) and the display of the views
// ([views] layer).
//
// Like [main3.ts], this client only calls the PHP server for three
// actions: [init-session], [authenticate-user] and [get-admindata].
// Once the tax data (rates, ceilings...) has been retrieved with
// [get-admindata], it is the [business] layer ([Business] class, the same
// [Business.ts] file as [main3.ts]) that calculates the tax in the
// browser, without any new network call. The list of simulations is also
// kept only in the browser (array [this.simulations]),
// exactly as [main3.ts] does with its local array [simulations]
// (see `simulations.push(...)` and `simulations.splice(...)`).
// -----------------------------------------------------------------------
class AppHtml {

  private dao: Dao;
  private views: Views;
  // [business] layer: only exists once the tax data has been retrieved
  // (after a successful authentication) - see [initBusiness]
  private business?: Business;
  // list of the simulations already calculated, kept in the browser
  private simulations: Simulation[] = [];

  constructor() {
    // axios configuration - identical to [main3.ts]
    axios.defaults.timeout = 5000;
    axios.defaults.baseURL = 'http://localhost/';
    axios.defaults.withCredentials = true;
    // [dao] and [views] layers
    this.dao = new Dao(axios);
    this.views = new Views(document.getElementById('app')!);
  }

  // entry point of the application
  async start(): Promise<void> {
    try {
      // init session - mandatory before any other action (see [main.php])
      const response = await this.dao.initSession();
      if (response.status !== 700) {
        this.views.authenticationView(this.formatError(response.response), {
          onSubmit: (user, password) => this.authenticate(user, password)
        });
        return;
      }
    } catch (error: any) {
      this.views.authenticationView(
        "Unable to contact the server (" + error.message + ")",
        { onSubmit: (user, password) => this.authenticate(user, password) }
      );
      return;
    }
    // session initialized: we display the authentication
    this.showAuthentication();
  }

  // --------------------------------------------------------------------
  // [authentication] view
  // --------------------------------------------------------------------
  private showAuthentication(error?: string): void {
    this.views.authenticationView(error, {
      onSubmit: (user, password) => this.authenticate(user, password)
    });
  }

  private async authenticate(user: string, password: string): Promise<void> {
    try {
      const response = await this.dao.authenticateUser(user, password);
      if (response.status === 200) {
        // successful authentication: as in [main3.ts], we now retrieve,
        // once and for all, the tax data
        // needed by the [business] layer
        await this.initBusiness();
      } else {
        // failure (wrong login/password, or missing parameters)
        this.showAuthentication(this.formatError(response.response));
      }
    } catch (error: any) {
      this.showAuthentication("Unable to contact the server (" + error.message + ")");
    }
  }

  // retrieves the tax administration data and instantiates the
  // [business] layer with this data - a single server call for the whole session,
  // exactly like [main3.ts]
  private async initBusiness(): Promise<void> {
    try {
      const response = await this.dao.getAdminData();
      if (response.status === 1000) {
        this.business = new Business(response.response);
        // new work session: we start again from an empty list
        this.simulations = [];
        this.showCalculation();
      } else {
        this.showAuthentication(this.formatError(response.response));
      }
    } catch (error: any) {
      this.showAuthentication("Unable to contact the server (" + error.message + ")");
    }
  }

  // --------------------------------------------------------------------
  // [tax calculation] view
  // --------------------------------------------------------------------
  private showCalculation(result?: Simulation, error?: string): void {
    this.views.calculationView(result, error, {
      onSubmit: (married, children, salary) => this.calculate(married, children, salary),
      onList: () => this.showList(),
      onEndSession: () => this.endSession()
    });
  }

  // tax calculation done entirely in the browser ([business] layer):
  // no call to the server, unlike the first version of this client
  private calculate(married: string, children: number, salary: number): void {
    // this.business has necessarily been initialized: we only reach the
    // [calculation] view after [initBusiness]
    const result = this.business!.calculateTax(married, children, salary);
    // the result returned by [business.calculateTax] only contains the
    // calculated values (tax, surcharge, discount, reduction, rate): we add to it
    // the parameters entered by the user, so that they can be
    // displayed again in the list of simulations
    const simulation: Simulation = { married: married, children: children, salary: salary, ...result };
    // storage in the browser (like `simulations.push(...)` in [main3.ts])
    this.simulations.push(simulation);
    this.showCalculation(simulation);
  }

  // --------------------------------------------------------------------
  // [list of simulations] view
  // --------------------------------------------------------------------
  // no more server calls here: the list is already entirely in
  // [this.simulations]
  private showList(error?: string): void {
    this.views.listView(this.simulations, error, {
      onDelete: (number) => this.deleteSimulation(number),
      onCalculation: () => this.showCalculation(),
      onEndSession: () => this.endSession()
    });
  }

  private deleteSimulation(number: number): void {
    // local deletion from the array [this.simulations], like
    // `simulations.splice(1, 1)` in [main3.ts] - still no server call
    this.simulations.splice(number, 1);
    this.showList();
  }

  // --------------------------------------------------------------------
  // end of session
  // --------------------------------------------------------------------
  private endSession(): void {
    // end of session on the client side only: we forget the [business] layer
    // (and therefore the tax data) and the simulations already calculated, then we
    // go back to the authentication screen. This client does not need to
    // call the server for this.
    this.business = undefined;
    this.simulations = [];
    this.showAuthentication();
  }

  // --------------------------------------------------------------------
  // formats the [response] field of an error response from the server,
  // which is either a string or an array of strings (see PHP controllers)
  // --------------------------------------------------------------------
  private formatError(response: unknown): string {
    if (Array.isArray(response)) {
      return response.join(' ; ');
    }
    return String(response);
  }
}
